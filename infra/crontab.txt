# conform-rgaa-agent crontab — installer via :
#   crontab -u agent /opt/conform-rgaa-agent/infra/crontab.txt
# Fuseau cron = UTC.
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# Audit RGAA 44 critères (rgaa-ia.fr / /opt/rgaa).
# 4×/jour 6h, 10h, 14h, 18h UTC × 25 = ~100 audits/jour.
# Source de vérité conformité : mairie_latest_audit (vue sur mairie_audits).
0 6,10,14,18 * * * /opt/conform-rgaa-agent/scripts/audit_25_daily.sh

# Scrape footer RGAA (regex/HTML, sans IA) — alimente rgaa_in_footer +
# rgaa_page_url (signaux légaux : déclaration publiée + lien dans le footer).
# Indépendant de l'audit technique. Batch 50, 1×/h 5h-21h UTC.
15 5-21 * * * cd /opt/conform-rgaa-agent && doppler run --project conform-rgaa --config dev -- python3 scripts/scrape_rgaa.py --batch 50 >> logs/scrape.log 2>&1

# Ingest API hebdomadaire (lundi 03h00 UTC) — refresh emails depuis api-lannuaire
0 3 * * 1 cd /opt/conform-rgaa-agent && doppler run --project conform-rgaa --config dev -- python3 scripts/ingest_api.py >> logs/ingest.log 2>&1

# 5 emails de prospection Mon-Fri 7h30 Paris (DST-aware via CRON_TZ).
# CRON_TZ s'applique aux lignes suivantes — gardé en fin de fichier.
CRON_TZ=Europe/Paris
30 7 * * 1-5 /opt/conform-rgaa-agent/scripts/send_5_daily.sh
